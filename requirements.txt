# SubAI Translator —— Python 后端依赖清单
# 安装：pip install -r requirements.txt
#
# 版本策略（[P2 2026-10-24]）：全部加上界，避免上游 major 升级在某天悄悄装坏。
# 历史踩坑：rapidocr-onnxruntime 曾误写 >=1.3（PyPI 实际只到 1.2.3，任何机器必然装失败）；
#          Pillow 曾因无下限带进已知漏洞（安全扫描 2026-09-29）。
#
# 三种安装口径：
#   1) 常规安装（推荐给使用者）：pip install -r requirements.txt
#   2) 精确复现（给排障/CI 相同环境用）：pip install -r requirements-lock.txt
#      —— lock 是本机 pip freeze 的全量快照（238 个 ==），跨平台/跨 Python 版本可能失败
#   3) 最小运行时（打包用）：只装「核心 + ASR + OCR」，跳过测试依赖
#
# 说明：
#   - 核心依赖（必装）支撑 API 服务、任务队列、缓存、翻译与字幕生成；
#   - ASR / OCR 为重型模型库，hardsub 与语音识别模式依赖它们；
#   - torch 并非必需：GPU 加速由 CTranslate2 + nvidia-* 运行库提供，torch 只在
#     nvidia-smi 不可用时作为显存探测的兜底（见 src/config.py 的 get_asr_device）。

# ---------------- 核心（必装）----------------

uvicorn[standard]>=0.29,<1
pydantic>=2.6,<3
pydantic-settings>=2.2,<3
httpx>=0.27,<1

# ---------------- ASR 语音识别 ----------------
faster-whisper>=1.0,<2

# sherpa-onnx：可选引擎（SenseVoice / Paraformer，中文更强、快 2~4 倍）。
# 不装也能跑（默认 engine=whisper）；装了但没模型时 pipeline 会给出可操作的报错。
# 为什么不用 funasr 库：它是训练框架，部署侧 VAD 行为不稳定（见 src/asr_sherpa.py 文件头）。
# 注意：sherpa-onnx 在 Python 3.14 上可正常安装（2.0.0 实测）。
sherpa-onnx>=1.13,<3

# ---------------- OCR 硬字幕识别 ----------------
# 注意：rapidocr-onnxruntime 在 PyPI 上最高只到 1.2.3（曾误写成 >=1.3，导致任何机器
# 执行 pip install -r requirements.txt 必然失败）；上界锁 1.3 防止下个大版本改 API。
rapidocr-onnxruntime>=1.2.3,<1.3
opencv-python>=4.8,<5
# 安全修复：Pillow < 12.3.0 存在多个已知漏洞（pip-audit 扫描结果，2026-09-29）
pillow>=12.3.0,<13

# ---------------- GPU 加速（可选，有 NVIDIA GPU 时启用）----------------
# torch>=2.1,<3

# ---------------- 测试 ----------------
pytest>=8.0,<9
pytest-asyncio>=0.23,<2

langdetect>=1.0.9,<2
